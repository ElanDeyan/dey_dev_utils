     280 /// Extension for chaining multiple validation rules on [Validated] results.
     281 ///
     282 /// This extension provides fluent API for applying sequential validation rules
     283 /// to a value. As you chain calls to [check], validation errors accumulate in
     284 /// a Set, allowing you to collect all failures at once rather than stopping
     285 /// at the first error.
     286 ///
     287 /// This is particularly useful for form validation where you want to show
     288 /// users all problems with their input simultaneously, improving user experience
     289 /// by avoiding multi-pass form submission.
     290 ///
     291 /// Example:
     292 /// ```dart
     293 /// // Chain multiple validation rules
     294 /// final email = check(
     295 ///   userInput,
     296 ///   (v) => v.isNotEmpty,
     297 ///   error: 'Email is required',
     298 /// )
     299 ///   .check((v) => v.contains('@'), error: 'Invalid email format')
     300 ///   .check((v) => !v.endsWith('.'), error: 'Cannot end with period')
     301 ///   .check((v) => v.length <= 100, error: 'Email too long');
     302 ///
     303 /// // Check if all rules passed
     304 /// if (email.isValid) {
     305 ///   final validEmail = email.unwrapOrNull()!;
     306 ///   saveEmail(validEmail);
     307 /// } else {
     308 ///   final invalid = email as Invalid<String>;
     309 ///   showErrors(invalid.errors); // Show all errors to user
     310 /// }
     311 /// ```
     312 @experimental
     313 extension ValidatedOps<T> on Validated<T> {
     314   /// Applies an additional validation rule to this result.
     315   ///
     316   /// Behavior depends on the current state:
     317   /// - If [Invalid]: Adds [error] to the existing error set and returns [Invalid]
     318   /// - If [Valid]: Evaluates [rule] on the value:
     319   ///   - If [rule] returns true: Returns this [Valid] unchanged
     320   ///   - If [rule] returns false: Returns [Invalid] with the [error]
     321   ///
     322   /// Errors accumulate in a Set, so calling check multiple times with the same
     323   /// [error] won't create duplicates. The original value is preserved through
     324   /// the entire chain.
     325   ///
     326   /// Parameters:
     327   /// - [rule]: Predicate function that returns true if valid, false if invalid
     328   /// - [error]: Error message to include if this validation fails
     329   ///
     330   /// Example:
     331   /// ```dart
     332   /// // Successful chain - all rules pass
     333   /// final result = Valid('Alice')
     334   ///   .check((n) => n.length >= 2, 'Too short')
     335   ///   .check((n) => n.length <= 20, 'Too long');
     336   /// // result is Valid('Alice')
     337   ///
     338   /// // Failed chain - collects all errors
     339   /// final result2 = Valid('A')
     340   ///   .check((n) => n.length >= 2, 'Too short')
     341   ///   .check((n) => n.length <= 1, 'Must be max 1 char');
     342   /// // result2 is Invalid({'Too short'})
     343   ///
     344   /// // Chaining on Invalid - errors keep accumulating
     345   /// final result3 = Invalid<String>({'Required'}).check(
     346   ///   (_) => false,
     347   ///   'Too short',
     348   /// );
     349   /// // result3 is Invalid({'Required', 'Too short'})
     350   /// ```
     351   Validated<T> check(bool Function(T) rule, {required String error}) =>
     352       switch (this) {
     353         Invalid(:final errors) => .invalid({...errors, error}),
     354         Valid(:final value) => rule(value) ? this : .invalid({error}),
     355       };
     356 }
