      32 /// A middleware function that wraps or intercepts a handler.
      33 ///
      34 /// A [Middleware] takes an existing [Handler] and returns a new [Handler]
      35 /// that provides additional behavior before or after calling the original handler.
      36 /// This enables composition of cross-cutting concerns like logging, validation,
      37 /// error handling, timing, etc.
      38 ///
      39 /// Middleware follows the decorator pattern:
      40 /// - It receives the "next" handler in the chain
      41 /// - It creates a new handler that wraps the next handler
      42 /// - When called, it can perform pre-processing, call the next handler, then post-process
      43 ///
      44 /// Example:
      45 /// ```dart
      46 /// // Timing middleware that logs how long the handler takes
      47 /// Middleware<String> timingMiddleware = (next) {
      48 ///   return (input) async {
      49 ///     final sw = Stopwatch()..start();
      50 ///     try {
      51 ///       return await next(input);
      52 ///     } finally {
      53 ///       sw.stop();
      54 ///       print('Took ${sw.elapsedMilliseconds}ms');
      55 ///     }
      56 ///   };
      57 /// };
      58 ///
      59 /// // Validation middleware that checks input before processing
      60 /// Middleware<String> validationMiddleware = (next) {
      61 ///   return (input) async {
      62 ///     if (input.isEmpty) throw ArgumentError('Input cannot be empty');
      63 ///     return await next(input);
      64 ///   };
      65 /// };
      66 /// ```

      70 /// A composable pipeline of middleware that processes values asynchronously.
      71 ///
      72 /// [Pipeline] builds and executes a chain of middleware around a terminal handler.
      73 /// Middleware are registered via [use] and executed in the order they were added
      74 /// when the pipeline is called.
      75 ///
      76 /// **Execution order:** When you call the pipeline, middleware execute in an
      77 /// "onion layer" pattern:
      78 /// 1. First middleware's pre-processing runs
      79 /// 2. Second middleware's pre-processing runs
      80 /// 3. ... and so on until reaching the core handler
      81 /// 4. The core handler executes
      82 /// 5. Then post-processing runs in reverse order (last to first)
      83 ///
      84 /// **Example execution trace:**
      85 /// ```dart
      86 /// final pipeline = Pipeline<String>();
      87 /// pipeline.use(mw1); // Added first
      88 /// pipeline.use(mw2); // Added second
      89 /// pipeline.use(mw3); // Added third
      90 ///
      91 /// pipeline('hello');
      92 /// // Execution order:
      93 /// // → mw1 pre-processing
      94 /// //   → mw2 pre-processing
      95 /// //     → mw3 pre-processing
      96 /// //       → core handler
      97 /// //     ← mw3 post-processing
      98 /// //   ← mw2 post-processing
      99 /// // ← mw1 post-processing
     100 /// ```
     101 ///
     102 /// This pattern is useful for:
     103 /// - **Logging and monitoring**: Track all values flowing through
     104 /// - **Error handling**: Wrap operations in try-catch
     105 /// - **Validation**: Check and validate inputs before core logic
     106 /// - **Performance timing**: Measure execution duration
     107 /// - **Authentication/Authorization**: Check permissions before processing
     108 /// - **Transformation**: Modify values before or after processing
     109 ///
     110 /// **Complete example:**
     111 /// ```dart
     112 /// // Create a pipeline for processing strings
     113 /// final pipeline = Pipeline<String>();
     114 ///
     115 /// // Add logging middleware
     116 /// pipeline.use((next) => (input) async {
     117 ///   print('Start: $input');
     118 ///   final result = await next(input);
     119 ///   print('End: $result');
     120 ///   return result;
     121 /// });
     122 ///
     123 /// // Add timing middleware
     124 /// pipeline.use((next) => (input) async {
     125 ///   final sw = Stopwatch()..start();
     126 ///   final result = await next(input);
     127 ///   sw.stop();
     128 ///   print('Duration: ${sw.elapsedMilliseconds}ms');
     129 ///   return result;
     130 /// });
     131 ///
     132 /// // Add transformation middleware
     133 /// pipeline.use((next) => (input) async {
     134 ///   final transformed = input.toUpperCase();
     135 ///   return await next(transformed);
     136 /// });
     137 ///
     138 /// // Call the pipeline
     139 /// final result = await pipeline('hello');
     140 /// // Logs:
     141 /// // Start: HELLO
     142 /// // Duration: 0ms
     143 /// // End: HELLO
     144 /// ```

     146 class Pipeline<T extends Object?> {
     147   /// Stores the list of middleware in the order they were registered.
     148   final List<Middleware<T>> _middlewares = [];

     150   /// Executes the pipeline with the given input.
     151   ///
     152   /// This method:
     153   /// 1. Builds the middleware chain in reverse order (right-to-left)
     154   /// 2. Each middleware wraps the previous handler
     155   /// 3. Calls the complete chain with the input
     156   ///
     157   /// The chain is built as: `mw1(mw2(mw3(...(core_handler))))`
     158   /// where middleware are ordered as they were added via [use].
     159   ///
     160   /// Returns a [Future] that resolves when the entire pipeline completes.
     161   /// If any middleware throws an exception, it propagates through the Future.
     162   ///
     163   /// Parameters:
     164   /// - [input]: The value to process through the pipeline
     165   ///
     166   /// Returns: A [Future] containing the processed value
     167   ///
     168   /// Example:
     169   /// ```dart
     170   /// final pipeline = Pipeline<int>();
     171   /// pipeline.use((next) => (value) async => await next(value * 2));
     172   /// final result = await pipeline(5); // Returns 10
     173   /// ```
     174   Future<T> call(T input) {
     175     var handler = _handler<T>;
     176 
     177     // Build chain right to left: first middleware ends up outermost
     178     for (final mw in _middlewares.reversed) {
     179       final next = handler;
     180       handler = mw(next);
     181     }
     182     return handler(input);
     183   }
     184 
     185   /// Registers a middleware in this pipeline.
     186   ///
     187   /// Middleware are executed in the order they are registered (first registered
     188   /// is outermost, wrapping all later middleware). This is the same behavior
     189   /// as popular web frameworks like Express.js.
     190   ///
     191   /// Each middleware receives the next handler in the chain and must eventually
     192   /// call it (unless intentionally short-circuiting the chain).
     193   ///
     194   /// Parameters:
     195   /// - [middleware]: The middleware function to add
     196   ///
     197   /// Example:
     198   /// ```dart
     199   /// pipeline.use((next) => (input) async {
     200   ///   print('Before');
     201   ///   final result = await next(input);
     202   ///   print('After');
     203   ///   return result;
     204   /// });
     205   /// ```
     206   void use(Middleware<T> middleware) => _middlewares.add(middleware);
