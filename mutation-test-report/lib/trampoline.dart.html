       1 /// Executes a bouncing computation until completion without recursion.
       2 ///
       3 /// The trampoline pattern prevents stack overflow by replacing recursive
       4 /// function calls with iterative execution. Instead of making recursive calls
       5 /// directly, return a [More] bounce with a thunk that will be executed next.
       6 ///
       7 /// This is particularly useful for:
       8 /// - Tree traversals that could overflow the stack
       9 /// - Deep recursive algorithms (parsing, DFS, etc.)
      10 /// - Tail-recursive patterns that can't be optimized by the compiler
      11 ///
      12 /// How it works:
      13 /// 1. Start with an initial [Bounce] (usually [More])
      14 /// 2. Each [More] returns a function that produces the next [Bounce]
      15 /// 3. The trampoline executes these functions in a loop (not recursion)
      16 /// 4. When [Done] is reached, the final value is returned
      17 ///
      18 /// Example:
      19 /// ```dart
      20 /// // Calculate factorial without recursion overflow
      21 /// Bounce<int> factorialBounce(int n, int acc) {
      22 ///   if (n <= 1) {
      23 ///     return Done(acc);
      24 ///   }
      25 ///   return More(() => factorialBounce(n - 1, n * acc));
      26 /// }
      27 ///
      28 /// final result = trampoline(More(() => factorialBounce(5, 1)));
      29 /// // result = 120
      30 /// ```
      31 T trampoline<T>(Bounce<T> bounce) {
      32   var current = bounce;
      33 
      34   while (current is More<T>) {
      35     current = current.next();
      36   }
      37 
      38   assert(current is Done<T>);
      39 
      40   return (current as Done<T>).value;
      41 }

      43 /// Base class for computations in the trampoline pattern.
      44 ///
      45 /// A [Bounce] represents either:
      46 /// - More computation to do ([More])
      47 /// - The final result ([Done])
      48 ///
      49 /// This sealed class restricts implementations to [More] and [Done],
      50 /// ensuring type safety in the trampoline execution.
      51 sealed class Bounce<T extends Object?> {
      52   const Bounce();
      53 
      54   const factory Bounce.done(T value) = Done<T>;
      55   const factory Bounce.more(Bounce<T> Function() action) = More<T>;
      56 }

      59 ///
      60 /// When the trampoline reaches a [Done] bounce, it extracts the [value]
      61 /// and returns it. This signals that the computation is complete.

      63 /// Example:
      64 /// ```dart
      65 /// final result = Done(42);
      66 /// // Trampoline will extract and return 42

      68 final class Done<T extends Object?> extends Bounce<T> {

      70   const Done(this.value);

      72   /// The final computed value that will be returned by the trampoline.

      76 /// Represents pending computation in the trampoline pattern.
      77 ///
      78 /// Each [More] contains a thunk (lazy function) that will be executed
      79 /// to produce the next [Bounce]. This defers execution and prevents
      80 /// direct recursion, allowing the trampoline loop to handle deep stacks.
      81 ///
      82 /// Example:
      83 /// ```dart
      84 /// // Represent pending recursive computation
      85 /// return More(() => nextBounce(n - 1));
      86 /// // Instead of: return nextBounce(n - 1);
      87 /// ```
      88 final class More<T extends Object?> extends Bounce<T> {

      90   ///
      91   /// The [next] function will be called by the trampoline to get the
      92   /// next bounce in the chain. This deferral is key to preventing
      93   /// stack overflow in deep recursion.
      94   const More(this.next);

      96   /// A lazy function that produces the next computation step.
      97   ///
      98   /// The trampoline calls this function to get the next [Bounce].
      99   /// Wrap recursive calls in this thunk to avoid direct recursion:

     101   /// ```dart
     102   /// // Good: deferred recursion
     103   /// return More(() => recursiveStep(n - 1));

     105   /// // Bad: would still be recursive

     107   /// ```
     108   final Bounce<T> Function() next;
