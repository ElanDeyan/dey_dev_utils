       7 /// An isomorphism: a reversible transformation between two types.
       8 ///
       9 /// An [Iso] represents a bidirectional mapping between types [A] and [B].
      10 /// It consists of two inverse functions:
      11 /// - [to]: transforms from [A] to [B]
      12 /// - [from]: transforms from [B] to [A]
      13 ///
      14 /// These functions form an isomorphism, meaning they are perfect inverses
      15 /// of each other: `from(to(a)) == a` and `to(from(b)) == b` for all valid
      16 /// values. This property is assumed but not enforced by the type system.
      17 ///
      18 /// **Use cases:**
      19 /// - Type conversions with guaranteed round-trip safety
      20 /// - Bidirectional serialization/deserialization
      21 /// - Format conversions (e.g., JSON ↔ Dart objects)
      22 /// - Domain modeling (e.g., string representations ↔ enums)
      23 ///
      24 /// **Example:**
      25 /// ```dart
      26 /// // Define an isomorphism between String and int
      27 /// final stringToInt = Iso(
      28 ///   to: (s) => int.parse(s),
      29 ///   from: (i) => i.toString(),
      30 /// );
      31 ///
      32 /// // Use it
      33 /// assert(stringToInt.to('42') == 42);
      34 /// assert(stringToInt.from(42) == '42');
      35 /// assert(stringToInt.from(stringToInt.to('42')) == '42'); // Round-trip
      36 /// ```
      37 @experimental
      38 final class Iso<A extends Object?, B extends Object?> {
      39   /// Creates an isomorphism with the given [to] and [from] transformations.
      40   ///
      41   /// Both [to] and [from] are required. For this to represent a valid
      42   /// isomorphism, they must be proper inverses of each other.
      43   const Iso({required this.to, required this.from});
      44 
      45   /// Transforms a value from type [A] to type [B].
      46   ///
      47   /// This should be the inverse of [from], such that
      48   /// `from(to(a)) == a` for all values [a].
      49   final B Function(A) to;
      50 
      51   /// Transforms a value from type [B] to type [A].
      52   ///
      53   /// This should be the inverse of [to], such that
      54   /// `to(from(b)) == b` for all values [b].
      55   final A Function(B) from;
      56 
      57   /// Downgrades this isomorphism to a [Lens].
      58   ///
      59   /// Every isomorphism is a lens (you can always get and set), but not every
      60   /// lens is an isomorphism. This returns a lens where:
      61   /// - **get** uses [to] to extract the [B] value from an [A]
      62   /// - **set** uses [from] to reconstruct the [A] value from a [B]
      63   ///
      64   /// The old value is ignored during set (since isomorphisms can fully
      65   /// reconstruct from just the new part).
      66   ///
      67   /// Example:
      68   /// ```dart
      69   /// final iso = Iso(to: (s) => int.parse(s), from: (i) => i.toString());
      70   /// final lens = iso.asLens;
      71   /// assert(lens.get('123') == 123);
      72   /// assert(lens.set('old', 456) == '456');
      73   /// ```
      74   ///
      75   /// See also: [Lens]
      76   Lens<A, B> get asLens => Lens(get: to, set: (_, b) => from(b));
      77 
      78   /// Flips the direction of this isomorphism.
      79   ///
      80   /// Returns a new [Iso] that transforms from [B] to [A] instead of [A] to [B],
      81   /// by swapping the [to] and [from] functions. This is mathematically free
      82   /// since isomorphisms are symmetric.
      83   ///
      84   /// If `iso.to(a) == b`, then `iso.inverse.from(b) == a`.
      85   ///
      86   /// Example:
      87   /// ```dart
      88   /// final iso = Iso(to: int.parse, from: (i) => i.toString());
      89   /// final inv = iso.inverse;
      90   /// assert(inv.to(42) == '42');
      91   /// assert(inv.from('42') == 42);
      92   /// ```
      93   Iso<B, A> get inverse => Iso(to: from, from: to);
      94 
      95   /// Lifts this isomorphism to work on lists of values.
      96   ///
      97   /// Returns a new [Iso] that transforms `List<A>` to `List<B>` (and vice versa)
      98   /// by applying [to] and [from] to each element. This is useful when you want
      99   /// to apply an isomorphism to a collection of values.
     100   ///
     101   /// If `iso` is a valid isomorphism, then `iso.isoList` is also a valid
     102   /// isomorphism (the round-trip property holds element-wise).
     103   ///
     104   /// Example:
     105   /// ```dart
     106   /// final iso = Iso(to: int.parse, from: (i) => i.toString());
     107   /// final listIso = iso.isoList;
     108   /// assert(listIso.to(['1', '2', '3']) == [1, 2, 3]);
     109   /// assert(listIso.from([10, 20]) == ['10', '20']);
     110   /// ```
     111   Iso<List<A>, List<B>> get isoList =>
     112       Iso(to: (as) => as.map(to).toList(), from: (bs) => bs.map(from).toList());
     113 
     114   /// Composes this isomorphism with another isomorphism.
     115   ///
     116   /// Chains two isomorphisms: `A ↔ B` and `B ↔ C` to create a new isomorphism
     117   /// `A ↔ C`. This allows you to build complex transformations by composing
     118   /// simpler ones.
     119   ///
     120   /// The composition follows mathematical function composition:
     121   /// - Forward: `other.to(to(a))` - apply this iso's [to], then the other's [to]
     122   /// - Backward: `from(other.from(c))` - apply the other's [from], then this iso's [from]
     123   ///
     124   /// Type parameter [C] is the final type in the chain.
     125   ///
     126   /// Example:
     127   /// ```dart
     128   /// // Chain: String -> int -> bool (if int.isEven)
     129   /// final toInt = Iso(to: int.parse, from: (i) => i.toString());
     130   /// final toBool = Iso(to: (i) => i.isEven, from: (b) => b ? 2 : 1);
     131   /// final stringToBool = toInt.then(toBool);
     132   ///
     133   /// assert(stringToBool.to('4') == true);   // '4' -> 4 -> true
     134   /// assert(stringToBool.from(false) == '1'); // false -> 1 -> '1'
     135   /// ```
     136   Iso<A, C> then<C>(Iso<B, C> other) =>
     137       Iso(to: (a) => other.to(to(a)), from: (c) => from(other.from(c)));
     138 }
